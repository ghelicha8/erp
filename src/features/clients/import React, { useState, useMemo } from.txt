                {client.creditScore} <Star className="w-3.5 h-3.5 fill-amber-500" />
              </div>
            </div>

            <div className="flex flex-col">
              <span className="text-[10px] font-black text-indigo-500 bg-indigo-50 dark:bg-indigo-900/30 px-2 py-1 rounded-md w-max mb-1 border border-indigo-100 dark:border-indigo-500/20">
                {client.type === 'PERSON' ? 'شخص حقیقی' : 'شخص حقوقی / شرکت'}
              </span>
              <h2 className="text-2xl font-black text-slate-800 dark:text-white drop-shadow-sm">
                {client.name} <span className="text-indigo-600 dark:text-indigo-400">{client.lastName}</span>
              </h2>
              <span className="text-sm font-bold text-slate-500 flex items-center gap-1 mt-1.5 opacity-90">
                <Phone className="w-4 h-4" /> {client.phone}

            </div>

            {/* کارت ۱: کیف پول */}
            <div className="bg-white/60 dark:bg-slate-800/60 backdrop-blur-md p-4 rounded-[1.5rem] border border-white/80 dark:border-slate-600 shadow-sm relative overflow-hidden group">

              <input type="text" placeholder="تا تاریخ..." value={dateTo} onChange={e => setDateTo(e.target.value)} className="w-full bg-transparent outline-none text-sm font-bold text-slate-700 dark:text-slate-200" />
            </div>
          </div>

          const Icon = tab.icon;
          return (
            <button
              key={tab.id}
              onClick={() => setActiveTab(tab.id)}
              className={`relative flex items-center gap-2 px-5 py-3 rounded-xl text-sm font-black transition-all shrink-0 overflow-hidden ${

              )}

              <span className="relative z-10">{tab.label}</span>
            </button>
          );
        })}
      </div>

      {/* ================= 4. محتوای داینامیک تب‌ها (مرحله بعدی) ================= */}
      <div className="w-full flex-1 bg-white/40 dark:bg-slate-900/40 backdrop-blur-2xl border border-white/60 dark:border-slate-700/60 shadow-xl rounded-[2.5rem] p-6 sm:p-8 z-[70] relative min-h-[400px]">
        
        {/* Placeholder - تو مرحله بعد کدهای واقعی تب مالی رو اینجا میاریم */}
        <AnimatePresence mode="wait">
          <motion.div 
            key={activeTab}
            initial={{ opacity: 0, y: 10 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -10 }}
            transition={{ duration: 0.2 }}
            className="flex flex-col items-center justify-center h-full text-slate-400 py-20"
          >
            <div className="w-24 h-24 bg-white/50 dark:bg-slate-800/50 rounded-[2rem] border-2 border-dashed border-slate-300 dark:border-slate-600 flex items-center justify-center mb-6">

            </div>

              این بخش در مرحله بعدی (طبق معماری عنکبوتی و استفاده از فیلترهای گلوبال بالا) با دقت ۱۰۰٪ طراحی و متصل می‌شود.
